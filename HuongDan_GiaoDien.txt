# Singleton Kafka Producer kết nối cổng 9092
_kafka_producer = None

def get_kafka_producer():
    global _kafka_producer
    if _kafka_producer is None and KafkaProducer is not None:
        try:
            _kafka_producer = KafkaProducer(
                bootstrap_servers="localhost:9092",
                value_serializer=lambda v: json.dumps(v).encode("utf-8"),
                request_timeout_ms=1000
            )
        except Exception:
            _kafka_producer = False
    return _kafka_producer if _kafka_producer else None

def send_kafka_event(movie_id: int, rating: float, action: str = "rate", user_id: int = 99999):
    producer = get_kafka_producer()
    if producer:
        try:
            event = {
                "timestamp": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                "user_id": int(user_id),
                "movie_id": int(movie_id),
                "action": str(action),
                "rating": float(rating)
            }
            producer.send("movie_logs", event)
            print(f"📡 [Kafka] Sent event to movie_logs: {event}")
            return True
        except Exception as e:
            print(f"⚠️ [Kafka] Send failed: {e}")
    return False

# --- API TÌM KIẾM PHIM THEO TÊN (AUTOCOMPLETE TRÊN TOÀN BỘ 62.423 PHIM) ---
@app.get("/api/search_movies")
def search_movies(q: str = Query(..., min_length=1)):
    try:
        df = load_movies_catalog()
        matches = df[df["TenPhim"].str.contains(q, case=False, na=False)].head(15)
        return matches[["movieId", "TenPhim", "TheLoai"]].to_dict(orient="records")
    except Exception as e:
        return []

# --- API GỢI Ý PHIM DỰA TRÊN PHIM ĐÃ CHỌN & SỐ ĐIỂM ĐÁNH GIÁ (ITEM-TO-ITEM TRÊN 100% KHO DỮ LIỆU) ---
@app.get("/api/recommend_by_movie")
def recommend_by_movie(movieId: int, rating: int = 5, limit: int = 12):
    try:
        import time
        t0 = time.time()
        df = load_movies_catalog()
        
        # Tìm phim mục tiêu
        target_series = df[df["movieId"] == movieId]
        if target_series.empty:
            target_genres = "Action|Adventure|Sci-Fi"
            target_title = "Phim đã chọn"
        else:
            target_genres = target_series.iloc[0]["TheLoai"]
            target_title = target_series.iloc[0]["TenPhim"]

        target_set = set(str(target_genres).split("|"))

        # Thuật toán tính độ tương đồng Jaccard trên TOÀN BỘ 62.423 phim trong kho
        candidate_df = df[df["movieId"] != movieId]
        weight = (rating / 5.0)

        # Tính toán vectorized/apply trên toàn bộ kho dữ liệu
        def compute_score(g_set):
            inter = len(target_set & g_set)
            if inter == 0:
                return 0.0
            union = len(target_set | g_set)
            jaccard = inter / union
            return round(min((jaccard * 0.75 + 0.25) * weight * 100, 99.4), 1)

        scores = candidate_df["genre_set"].apply(compute_score)
        matched_idx = scores[scores > 0].index
        
        top_matches = candidate_df.loc[matched_idx].copy()
        top_matches["match"] = scores.loc[matched_idx]
        top_matches.sort_values(by="match", ascending=False, inplace=True)
        top_results = top_matches.head(limit)

        duration = round(time.time() - t0, 3)

        # Gửi log tương tác người dùng vào Kafka topic movie_logs
        kafka_sent = send_kafka_event(movie_id=movieId, rating=rating, action="rate")

        return {
            "total_scanned": len(df),
            "duration_sec": duration,
            "kafka_sent": kafka_sent,
            "target": {"movieId": movieId, "TenPhim": target_title, "TheLoai": target_genres, "rating": rating},
            "movies": top_results[["movieId", "TenPhim", "TheLoai", "match"]].to_dict(orient="records")
        }
    except Exception as e:
        return {"error": str(e), "movies": [], "total_scanned": 0, "kafka_sent": False}
